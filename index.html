🔐 Complete Supabase SQL Schema & Row Level Security (RLS)

Copy and run this inside your Supabase SQL Editor. It enforces strict public read-only access and admin-only writes.

-- ============================================================================
-- EVENT RENTAL BUSINESS SCHEMA + ROW LEVEL SECURITY (SUPABASE POSTGRESQL)
-- ============================================================================

-- 1. Categories Table
CREATE TABLE IF NOT EXISTS public.categories (
  id TEXT PRIMARY KEY,
  name_en TEXT NOT NULL,
  name_bn TEXT DEFAULT '',
  display_order INTEGER NOT NULL DEFAULT 1,
  is_hidden BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Products Table
CREATE TABLE IF NOT EXISTS public.products (
  id TEXT PRIMARY KEY,
  sku TEXT DEFAULT '',
  name_en TEXT NOT NULL,
  name_bn TEXT DEFAULT '',
  category_id TEXT REFERENCES public.categories(id) ON DELETE RESTRICT,
  description TEXT DEFAULT '',
  price NUMERIC(10, 2) DEFAULT NULL,
  pricing_unit TEXT NOT NULL DEFAULT 'per day',
  pricing_notes TEXT DEFAULT '',
  stock_quantity INTEGER DEFAULT NULL,
  images JSONB DEFAULT '[]'::jsonb,
  is_published BOOLEAN NOT NULL DEFAULT TRUE,
  is_available BOOLEAN NOT NULL DEFAULT TRUE,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Business Settings Table (Single-row configuration)
CREATE TABLE IF NOT EXISTS public.business_settings (
  id INTEGER PRIMARY KEY CHECK (id = 1),
  business_name TEXT NOT NULL,
  logo_text TEXT DEFAULT 'ER',
  tagline TEXT DEFAULT '',
  phone_number TEXT DEFAULT '',
  whatsapp_number TEXT DEFAULT '',
  hero_title TEXT DEFAULT '',
  hero_subtext TEXT DEFAULT '',
  service_area TEXT DEFAULT '',
  opening_hours TEXT DEFAULT '',
  collection_instructions TEXT DEFAULT '',
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Enable Row Level Security (RLS) on all tables
ALTER TABLE public.categories ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.products ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.business_settings ENABLE ROW LEVEL SECURITY;

-- 5. RLS Policies: Public can ONLY read published/visible records
CREATE POLICY "Public can read visible categories"
  ON public.categories FOR SELECT
  USING (is_hidden = FALSE OR auth.role() = 'authenticated');

CREATE POLICY "Public can read published products"
  ON public.products FOR SELECT
  USING (is_published = TRUE OR auth.role() = 'authenticated');

CREATE POLICY "Public can read business settings"
  ON public.business_settings FOR SELECT
  USING (TRUE);

-- 6. RLS Policies: ONLY Authenticated Admin can Insert/Update/Delete
CREATE POLICY "Authenticated admin full access categories"
  ON public.categories FOR ALL
  TO authenticated
  USING (auth.role() = 'authenticated')
  WITH CHECK (auth.role() = 'authenticated');

CREATE POLICY "Authenticated admin full access products"
  ON public.products FOR ALL
  TO authenticated
  USING (auth.role() = 'authenticated')
  WITH CHECK (auth.role() = 'authenticated');

CREATE POLICY "Authenticated admin full access settings"
  ON public.business_settings FOR ALL
  TO authenticated
  USING (auth.role() = 'authenticated')
  WITH CHECK (auth.role() = 'authenticated');

-- 7. Storage Bucket for Product Photos (Run in SQL Editor)
INSERT INTO storage.buckets (id, name, public)
VALUES ('product-images', 'product-images', true)
ON CONFLICT (id) DO NOTHING;

CREATE POLICY "Public read access for product images"
  ON storage.objects FOR SELECT
  USING (bucket_id = 'product-images');

CREATE POLICY "Admin upload/delete product images"
  ON storage.objects FOR ALL
  TO authenticated
  USING (bucket_id = 'product-images' AND auth.role() = 'authenticated')
  WITH CHECK (bucket_id = 'product-images' AND auth.role() = 'authenticated');

🚀 Step-by-Step Deployment Instructions (Windows + GitHub + Cloudflare Pages)

  1. Create Free Supabase Account: Go to supabase.com, create a new project, open the SQL Editor, paste the script above, and click Run.
  2. Create Your Admin User & Disable Public Signups: In Supabase → Authentication → Users, click Add User → Create New User with your admin email & strong password. Then go to Authentication → Providers → Email and toggle OFF "Allow new users to sign up" so nobody else can register.
  3. Save Code on Windows: Save this file as index.html in a folder on your computer (e.g., C:\rental-website\index.html).
  4. Push to GitHub: Create a free repository on github.com, upload index.html, and commit.
  5. Deploy Free on Cloudflare Pages: Log in to dash.cloudflare.com → Workers & Pages → Create → Pages → Connect to Git. Select your repository and click Save and Deploy. No build command is needed. Your site will be live globally on https://your-project.pages.dev in 60 seconds!
